Frontend data modeltypes.ts
type Venue = 'SIG' | 'KALSHI' | 'POLYMARKET';
type Mode  = 'research' | 'paper' | 'shadow'
           | 'human_confirmed' | 'constrained_live';
type Dec   = string;   // decimal string: prices, money. never float

interface Stamp {
  exchangeTs: string | null; receivedTs: string;
  ageMs: number; stale: boolean;     // stale data is flagged, never dropped
  source: 'live' | 'sqlite' | 'fixture';
}
interface Level { price: Dec; size: number }
interface Book {
  venue: Venue; venueMarketId: string; outcome: 'YES' | 'NO';
  bids: Level[]; asks: Level[];      // as published; no imputed levels
  visibleDepthOnly: true; stamp: Stamp;
}
interface Mapping {
  id: string; sigMarketId: number; venue: Venue; venueMarketId: string;
  status: 'pending' | 'approved' | 'rejected';
  reviewer: string | null; reviewedAt: string | null;
  settlement: SettlementAttr[];
}
interface SettlementAttr {
  attribute: string;
  excerpt: Partial<Record<Venue, string | null>>;
  ruleSha: Partial<Record<Venue, string | null>>;
  review: 'unreviewed' | 'match' | 'differs' | 'unknown';
}
interface Fee { amount: Dec | null; status: 'known' | 'unknown';
              scheduleVersion: string | null }
interface Leg {
  venue: Venue; venueMarketId: string; outcome: 'YES' | 'NO';
  side: 'BUY' | 'SELL'; limit: Dec; qty: number;
  vwap: Dec | null; availableQty: number; fee: Fee; book: Stamp;
}
// the single gate list: dashboard, ticket and /confirm all use it
type GateId = 'sig_auth' | 'payload_verified' | 'venue_ids'
  | 'fees_known' | 'mapping_approved' | 'settlement_approved'
  | 'quotes_fresh' | 'liquidity' | 'no_breaker' | 'risk_limits'
  | 'recon_clean' | 'kill_switch_off' | 'live_mode';
interface Gate { id: GateId; pass: boolean; detail: string; at: string }
interface Block { code: string; message: string; gate: GateId | null }

interface Opportunity {
  id: string;
  strategy: 'sig_arb' | 'complete_set' | 'xv_move' | 'rel_value';
  event: string; legs: Leg[];
  grossEdge: Dec; netRoi: Dec | null; capital: Dec | null;
  execQty: number; mappingIds: string[];
  news: 'clear' | 'watch' | 'breaker';
  risk: 'low' | 'med' | 'high';
  state: 'research_only' | 'blocked' | 'paper_eligible'
       | 'not_ready' | 'exec_ready';          // server-computed
  gates: Gate[]; blocks: Block[];
}
type OrderState = 'reviewed' | 'submitted' | 'acknowledged'
  | 'partially_filled' | 'filled' | 'cancelled' | 'rejected'
  | 'residual' | 'reconciled';
interface Fill { ts: string; qty: number; price: Dec; fee: Dec;
               venueOrderId: string }
interface TicketLeg extends Leg {
  clientOrderId: string; state: OrderState; fills: Fill[];
  breakEven: Dec | null; worstAcceptable: Dec; residualQty: number;
}
interface Ticket {
  id: string; opportunityId: string; mode: Mode;
  step: 'review' | 'confirm'; legs: TicketLeg[];
  quoteExpiresAt: string; gateHash: string;
  payload: object; payloadVerified: boolean;
  events: { ts: string; state: OrderState; leg: number;
            actor: 'operator' | 'system' | 'venue' }[];
}
interface SystemStatus {
  health: 'ok' | 'degraded' | 'down'; lastGoodRefresh: string;
  db: { ok: boolean; path: string; lastWrite: string };
  sigAuth: { session: boolean; payloadVerified: boolean };
  killSwitch: { engaged: boolean; by: string; at: string };
  mode: Mode; gates: Gate[]; feeds: (Stamp & { name: string })[];
}
